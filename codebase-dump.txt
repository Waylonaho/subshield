// Parse YYYY-MM-DD in local time to prevent UTC midnight off-by-one day shifts
function parseLocalDate(dateStr: string): Date | null {
  if (!dateStr) return null;
  const parts = dateStr.split("-").map(Number);
  if (parts.length !== 3 || parts.some(isNaN)) return null;
  return new Date(parts[0], parts[1] - 1, parts[2]);
}

function formatIsoDate(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

        currentStateRule.noticeTrigger === "last_furnishing"

        baseDate.setDate(
          baseDate.getDate() + currentStateRule.preliminaryNoticeDays
        );
        pDate = baseDate;
      }
    }

    if (lastWorkDate && currentStateRule?.lienFilingDays) {
      const baseLastDate = parseLocalDate(lastWorkDate);
      if (baseLastDate) {
        baseLastDate.setDate(
          baseLastDate.getDate() + currentStateRule.lienFilingDays
        );
        lDate = baseLastDate;
      }

function slugifyState(name: string): string {
  return name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "");
}

function getRuleBySlug(slug: string): StateLienRule | undefined {
  return Object.values(STATE_LIEN_RULES).find(
    (rule) => slugifyState(rule.name) === slug.toLowerCase()
  );
}

export async function generateStaticParams() {
  return Object.values(STATE_LIEN_RULES).map((rule) => ({
    state: slugifyState(rule.name),
  }));
}

export async function generateMetadata({
  params,
}: {
  params: Promise<{ state: string }>;
}): Promise<Metadata> {
  const { state } = await params;
  const rule = getRuleBySlug(state);

  if (!rule) {
    return { title: "State Lien Deadlines Not Found" };
  }

  const slug = slugifyState(rule.name);
  const title = `${rule.name} Mechanics Lien Deadline Calculator & Notice Rules (${rule.code})`;
  const description = `Calculate ${rule.name} preliminary notice and mechanics lien filing deadlines (${rule.lienFilingDays}-day statutory window under ${rule.statutoryNoticeName}). Free subcontractor calculator & .ics reminders.`;

  return {
    title,
    description,
    alternates: {
      canonical: `/lien-deadlines/${slug}`,
    },
    openGraph: {
      title,
      description,
      url: `https://www.subshieldhq.com/lien-deadlines/${slug}`,
      siteName: "SubShieldHQ",
      type: "website",
    },
  };
}

const STRIPE_PRO_URL = "https://buy.stripe.com/28E14n6FPd3fgpK8K52VG00";

function parseLocalDate(dateStr: string): Date | null {
  if (!dateStr) return null;
  const parts = dateStr.split("-").map(Number);
  if (parts.length !== 3 || parts.some(isNaN)) return null;
  return new Date(parts[0], parts[1] - 1, parts[2]);
}

function formatIsoDate(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

export default function StateLienCalculator({ rule }: { rule: StateLienRule }) {
  const [projectName, setProjectName] = useState<string>(`${rule.name} Commercial Job`);
  const [firstWorkDate, setFirstWorkDate] = useState<string>("");
  const [lastWorkDate, setLastWorkDate] = useState<string>("");

  const { prelimDate, lienDate } = useMemo(() => {
    let pDate: Date | null = null;
    let lDate: Date | null = null;

    if (rule.preliminaryNoticeDays) {
      const triggerBaseStr =
        rule.noticeTrigger === "last_furnishing" ? lastWorkDate : firstWorkDate;
      const baseDate = parseLocalDate(triggerBaseStr);
      if (baseDate) {
        baseDate.setDate(baseDate.getDate() + rule.preliminaryNoticeDays);
        pDate = baseDate;
      }
    }

    if (lastWorkDate && rule.lienFilingDays) {
      const baseLastDate = parseLocalDate(lastWorkDate);
      if (baseLastDate) {
        baseLastDate.setDate(baseLastDate.getDate() + rule.lienFilingDays);
        lDate = baseLastDate;
      }
    }

    return { prelimDate: pDate, lienDate: lDate };
  }, [rule, firstWorkDate, lastWorkDate]);

  const handleDownloadCalendar = () => {
    if (!prelimDate && !lienDate) {
      alert("Please enter your project work dates first to calculate deadlines.");
      return;
    }

    const icsString = generateIcsFile(
      projectName || `${rule.name} Project`,
      prelimDate ? formatIsoDate(prelimDate) : null,
      lienDate ? formatIsoDate(lienDate) : ""
    );

    const blob = new Blob([icsString], { type: "text/calendar;charset=utf-8" });
    const url = window.URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.setAttribute("download", `${rule.code}_SubShield_Lien_Deadlines.ics`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    window.URL.revokeObjectURL(url);
  };

  const renderPrelimDisplay = () => {
    if (!rule.preliminaryNoticeDays) {
      return "Not Statutorily Mandated";
    }
    if (prelimDate) {
      return prelimDate.toLocaleDateString("en-US", {
        month: "short",
        day: "numeric",
        year: "numeric",
      });
    }
    if (rule.noticeTrigger === "last_furnishing") {
      return "Enter Last Work Date";
    }
    if (rule.noticeTrigger === "invoice_date") {
      return "Enter First/Invoice Date";
    }
    return "Enter First Work Date";
  };

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6 sm:p-8 shadow-xl space-y-6">
      <div className="border-b border-slate-800 pb-4 flex flex-wrap items-center justify-between gap-2">
        <div>
          <h2 className="text-xl font-bold text-white">
            {rule.name} ({rule.code}) Mechanics Lien Deadline Calculator
          </h2>
          <p className="text-xs text-slate-400 mt-1">
            Statute Reference: {rule.statutoryNoticeName}
          </p>
        </div>
        <span className="text-xs font-semibold px-3 py-1 rounded-full bg-amber-500/10 text-amber-400 border border-amber-500/20">
          {rule.lienFilingDays}-Day Filing Window
        </span>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
        <div>
          <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-2">
            Project / Job Name

          <input

          />

        <div>

            First Day of Work / Delivery

          <input

export function generateIcsFile(
  projectName: string,
  preliminaryDeadline: string | null,
  lienDeadline: string
): string {
  const formatDate = (dateStr: string) => dateStr.replace(/-/g, '');

  let events = '';

  if (preliminaryDeadline) {
    const pDate = formatDate(preliminaryDeadline);
    events += `
BEGIN:VEVENT
UID:prelim-${Date.now()}@subshield.com
DTSTAMP:${formatDate(new Date().toISOString().slice(0, 10))}T090000Z
DTSTART;VALUE=DATE:${pDate}
SUMMARY:CRITICAL: Preliminary Notice Deadline - ${projectName}
DESCRIPTION:Statutory cutoff window to deliver preliminary notice or notice to owner. Missing this deadline forfeits mechanic's lien rights.
BEGIN:VALARM
ACTION:DISPLAY
DESCRIPTION:Preliminary Notice Deadline in 5 Days!
TRIGGER:-P5D
END:VALARM
END:VEVENT`;
  }

  const lDate = formatDate(lienDeadline);
  events += `
BEGIN:VEVENT
UID:lien-${Date.now()}@subshield.com
DTSTAMP:${formatDate(new Date().toISOString().slice(0, 10))}T090000Z
DTSTART;VALUE=DATE:${lDate}
SUMMARY:FINAL CUTOFF: Mechanics Lien Filing Deadline - ${projectName}
DESCRIPTION:Last statutory date to record a formal mechanic's lien against the real property parcel.
BEGIN:VALARM
ACTION:DISPLAY
DESCRIPTION:Lien Filing Cutoff in 14 Days!
TRIGGER:-P14D
END:VALARM
END:VEVENT`;

  return `BEGIN:VCALENDAR
VERSION:2.0
PRODID:-//SubShield HQ//Lien Safe Calendar//EN
CALSCALE:GREGORIAN
METHOD:PUBLISH${events}
END:VCALENDAR`.trim();
}
