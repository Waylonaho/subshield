function slugifyState(name: string): string {
  return name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "");
}

function parseLocalDate(dateStr: string): Date | null {
  if (!dateStr) return null;
  const parts = dateStr.split("-").map(Number);
  if (parts.length !== 3 || parts.some(isNaN)) return null;
  return new Date(parts[0], parts[1] - 1, parts[2]);
}

function formatIsoDate(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

export async function POST(req: NextRequest) {
  try {
    const authHeader = req.headers.get("authorization");
    const token = authHeader?.replace(/^Bearer\s+/i, "").trim();

    if (!token) {
      return NextResponse.json(
        { error: "Unauthorized. Please log in again." },
        { status: 401 }
      );
    }

    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
    const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
    const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY;
    const stripeSecretKey = process.env.STRIPE_SECRET_KEY;

    if (!supabaseUrl || !supabaseAnonKey || !serviceRoleKey) {
      return NextResponse.json(
        { error: "Server Supabase configuration missing." },
        { status: 500 }
      );
    }

    if (!stripeSecretKey) {
      return NextResponse.json(
        { error: "STRIPE_SECRET_KEY is not configured in environment variables." },
        { status: 500 }
      );
    }

    // Verify user session JWT
    const supabaseAuth = createClient(supabaseUrl, supabaseAnonKey);
    const {
      data: { user },
      error: authError,
    } = await supabaseAuth.auth.getUser(token);

    if (authError || !user?.email) {
      return NextResponse.json(
        { error: "Invalid or expired session." },
        { status: 401 }
      );
    }

    // Fetch Stripe customer ID via service role
    const supabaseAdmin = createClient(supabaseUrl, serviceRoleKey);
    const { data: subRecord, error: subError } = await supabaseAdmin
      .from("subscriptions")
      .select("stripe_customer_id")
      .eq("email", user.email.toLowerCase())
      .maybeSingle();

    if (subError || !subRecord?.stripe_customer_id) {
      return NextResponse.json(
        {
          error:
            "No Stripe customer record found for this account. Contact support@subshieldhq.com for assistance.",
        },
        { status: 404 }
      );
    }

    const origin =
      req.headers.get("origin") || "https://www.subshieldhq.com";

    // Create Stripe Billing Portal session via direct REST API call
    const formBody = new URLSearchParams({
      customer: subRecord.stripe_customer_id,
      return_url: `${origin}/dashboard`,
    });

    const stripeRes = await fetch(
      "https://api.stripe.com/v1/billing_portal/sessions",
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${stripeSecretKey}`,
          "Content-Type": "application/x-www-form-urlencoded",
        },
        body: formBody.toString(),
      }
    );

    const stripeData = await stripeRes.json();

    if (!stripeRes.ok || !stripeData.url) {
      return NextResponse.json(
        {
          error:
            stripeData?.error?.message ||
            "Failed to create Stripe billing portal session.",
        },
        { status: stripeRes.status || 500 }
      );
    }

    return NextResponse.json({ url: stripeData.url });
  } catch (err: unknown) {
    const message =
      err instanceof Error
        ? err.message
        : "Failed to create billing portal session.";
    return NextResponse.json({ error: message }, { status: 500 });
  }
}

function diffDaysFromToday(isoDateStr: string): number {
  const [y, m, d] = isoDateStr.split("-").map(Number);
  const target = new Date(y, m - 1, d);
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  return Math.round((target.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
}

export async function GET(req: Request) {
  const authHeader = req.headers.get("authorization");
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  // 1. Fetch active Pro subscriber emails
  const { data: activeSubs } = await supabaseAdmin
    .from("subscriptions")
    .select("email")
    .eq("status", "active");

  if (!activeSubs || activeSubs.length === 0) {
    return NextResponse.json({ sent: 0, message: "No active subscribers" });
  }

  const proEmails = new Set(activeSubs.map((s) => s.email.toLowerCase()));

  // 2. Fetch tracked projects
  const { data: projects } = await supabaseAdmin.from("projects").select("*");
  if (!projects) {
    return NextResponse.json({ sent: 0 });
  }

  const ALERT_WINDOWS = new Set([14, 7, 2]);
  let emailsSent = 0;

  for (const job of projects) {
    if (!proEmails.has(job.user_email.toLowerCase())) continue;

    // Check Preliminary Notice Window
    if (job.prelim_deadline) {
      const prelimDays = diffDaysFromToday(job.prelim_deadline);
      if (ALERT_WINDOWS.has(prelimDays)) {
        await resend.emails.send({
          from: "SubShield Alerts <alerts@subshieldhq.com>",
          to: job.user_email,
          subject: `⚠️ ${prelimDays}-Day Warning: ${job.state_code} Preliminary Notice Due (${job.project_name})`,
          text: `Statutory Alert for ${job.project_name} (${job.state_code}):\n\nYour Preliminary Notice deadline is in ${prelimDays} days (${job.prelim_deadline}).\n\nDeliver your notice via USPS Certified Mail before this cutoff expires to preserve your mechanics lien rights.\n\nManage your jobs: https://www.subshieldhq.com/dashboard`,
        });
        emailsSent++;
      }
    }

    // Check Final Lien Filing Cutoff
    if (job.lien_deadline) {
      const lienDays = diffDaysFromToday(job.lien_deadline);
      if (ALERT_WINDOWS.has(lienDays)) {
        await resend.emails.send({
          from: "SubShield Alerts <alerts@subshieldhq.com>",
          to: job.user_email,
          subject: `🚨 ${lienDays}-Day Cutoff: ${job.state_code} Mechanics Lien Deadline (${job.project_name})`,
          text: `CRITICAL STATUTORY CUTOFF for ${job.project_name} (${job.state_code}):\n\nYour final deadline to record a Mechanics Lien is in ${lienDays} days (${job.lien_deadline}).\n\nGenerate a Notice of Intent to Lien immediately: https://www.subshieldhq.com/tools/notice-of-intent-to-lien\nManage your dashboard: https://www.subshieldhq.com/dashboard`,
        });
        emailsSent++;
      }
    }
  }

  return NextResponse.json({ success: true, emailsSent });
}

import { NextResponse } from "next/server";
import crypto from "crypto";
import { createClient } from "@supabase/supabase-js";

const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

function verifyStripeSignature(
  rawBody: string,
  signatureHeader: string,
  secret: string
): boolean {
  const parts = signatureHeader.split(",");
  const timestamp = parts.find((p) => p.startsWith("t="))?.split("=")[1];
  const v1Signature = parts.find((p) => p.startsWith("v1="))?.split("=")[1];

  if (!timestamp || !v1Signature) return false;

  const signedPayload = `${timestamp}.${rawBody}`;
  const expectedSig = crypto
    .createHmac("sha256", secret)
    .update(signedPayload, "utf8")
    .digest("hex");

  try {
    return crypto.timingSafeEqual(
      Buffer.from(v1Signature),
      Buffer.from(expectedSig)
    );
  } catch {
    return false;
  }
}

export async function POST(req: Request) {
  const rawBody = await req.text();
  const signature = req.headers.get("stripe-signature") || "";
  const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET || "";

  if (!verifyStripeSignature(rawBody, signature, webhookSecret)) {
    return NextResponse.json({ error: "Invalid signature" }, { status: 400 });
  }

  const event = JSON.parse(rawBody);

  try {
    if (event.type === "checkout.session.completed") {
      const session = event.data.object;
      const email = (
        session.customer_details?.email ||
        session.customer_email ||
        ""
      )
        .toLowerCase()
        .trim();

      if (email) {
        await supabaseAdmin.from("subscriptions").upsert({
          email,
          stripe_customer_id: session.customer || null,
          status: "active",
          updated_at: new Date().toISOString(),
        });
      }
    } else if (
      event.type === "customer.subscription.updated" ||
      event.type === "customer.subscription.deleted"
    ) {
      const subscription = event.data.object;
      const status =
        subscription.status === "active" || subscription.status === "trialing"
          ? "active"
          : "inactive";

      if (subscription.customer) {
        await supabaseAdmin
          .from("subscriptions")
          .update({ status, updated_at: new Date().toISOString() })
          .eq("stripe_customer_id", subscription.customer);
      }
    }

    return NextResponse.json({ received: true });
  } catch (err) {
    return NextResponse.json({ error: "Webhook handler error" }, { status: 500 });
  }
}

import { STATE_LIEN_RULES } from "@/lib/statutoryRules";

const STRIPE_PRO_URL = "https://buy.stripe.com/28E14n6FPd3fgpK8K52VG00";

interface TrackedProject {
  id: string;
  project_name: string;
  state_code: string;
  first_work_date: string | null;
  last_work_date: string;
  prelim_deadline: string | null;
  lien_deadline: string;
}

function parseLocalDate(dateStr: string): Date | null {
  if (!dateStr) return null;
  const parts = dateStr.split("-").map(Number);
  if (parts.length !== 3 || parts.some(isNaN)) return null;
  return new Date(parts[0], parts[1] - 1, parts[2]);
}

function formatIsoDate(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

  const [loadingAuth, setLoadingAuth] = useState(true);
  const [loginEmail, setLoginEmail] = useState("");
  const [magicLinkSent, setMagicLinkSent] = useState(false);

  const [projects, setProjects] = useState<TrackedProject[]>([]);

  const [openingPortal, setOpeningPortal] = useState(false);

        a.name.localeCompare(b.name)
      ),
    []

    supabase.auth.getSession().then(({ data: { session } }) => {
      setUser(session?.user ?? null);
      setLoadingAuth(false);
    });

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setUser(session?.user ?? null);
    });

    return () => subscription.unsubscribe();
  }, []);

  useEffect(() => {
    if (!user?.email) return;

    const loadAccountData = async () => {
      const { data: sub } = await supabase
        .from("subscriptions")
        .select("status")
        .eq("email", user.email!.toLowerCase())
        .maybeSingle();

      setIsPro(sub?.status === "active");

      const { data: userProjects } = await supabase
        .from("projects")
        .select("*")
        .order("lien_deadline", { ascending: true });

      if (userProjects) setProjects(userProjects);
    };

    loadAccountData();

    e.preventDefault();
    const { error } = await supabase.auth.signInWithOtp({
      email: loginEmail.trim(),
      options: {
        emailRedirectTo: `${window.location.origin}/dashboard`,
      },
    });
    if (error) {
      alert(error.message);
    } else {
      setMagicLinkSent(true);
    }
  };

  const handleManageBilling = async () => {
    setOpeningPortal(true);

        data: { session },

    e.preventDefault();
    if (!user || !user.email) return;
    if (projects.length >= 30) {
      alert("Pro accounts support up to 30 active projects.");
      return;
    }

    const rule = STATE_LIEN_RULES[stateCode];
    if (!rule || !lastWorkDate) return;

    let prelimStr: string | null = null;
    if (rule.preliminaryNoticeDays) {
      const baseStr =
        rule.noticeTrigger === "last_furnishing" ? lastWorkDate : firstWorkDate;
      const baseDate = parseLocalDate(baseStr);
      if (baseDate) {
        baseDate.setDate(baseDate.getDate() + rule.preliminaryNoticeDays);
        prelimStr = formatIsoDate(baseDate);
      }
    }

    const lastDateObj = parseLocalDate(lastWorkDate);
    if (!lastDateObj) return;
    lastDateObj.setDate(lastDateObj.getDate() + rule.lienFilingDays);
    const lienStr = formatIsoDate(lastDateObj);

    setSavingProject(true);

      .from("projects")
      .insert({
        user_id: user.id,
        user_email: user.email.toLowerCase(),
        project_name: projectName.trim(),
        state_code: stateCode,
        first_work_date: firstWorkDate || null,
        last_work_date: lastWorkDate,
        prelim_deadline: prelimStr,
        lien_deadline: lienStr,
      })
      .select()
      .single();

    setSavingProject(false);
    if (error) {
      alert(error.message);
    } else if (data) {
      setProjects((prev) =>
        [...prev, data].sort((a, b) =>
          a.lien_deadline.localeCompare(b.lien_deadline)
        )
      );
      setProjectName("");
      setFirstWorkDate("");
      setLastWorkDate("");
    }
  };

  const handleDeleteProject = async (id: string) => {
    await supabase.from("projects").delete().eq("id", id);
    setProjects((prev) => prev.filter((p) => p.id !== id));
  };

  const daysRemaining = (dateStr: string) => {
    const target = parseLocalDate(dateStr);
    if (!target) return 0;
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    return Math.ceil((target.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
  };

  if (loadingAuth) {
    return (
      <div className="min-h-screen bg-slate-950 text-slate-300 flex items-center justify-center">
        Loading SubShield Pro Dashboard...
      </div>
    );

        <div className="max-w-6xl mx-auto px-4 h-16 flex items-center justify-between">

            Sub<span className="text-amber-500">Shield</span>HQ
          </Link>

                type="button"

                className="px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-amber-400 border border-slate-700 font-semibold transition cursor-pointer disabled:opacity-50"
              >
                {openingPortal ? "Opening..." : "Manage Billing / Cancel"}
              </button>
            )}

              <button
                type="button"
                onClick={() => supabase.auth.signOut()}
                className="px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-slate-300 border border-slate-700 cursor-pointer"
              >
                Sign Out
              </button>
            )}
          </div>
        </div>
      </header>

      <main className="max-w-5xl mx-auto px-4 py-12 flex-1 w-full">
        {!user ? (
          <div className="max-w-md mx-auto bg-slate-900 border border-slate-800 rounded-2xl p-8 space-y-5">
            <span className="text-xs font-bold uppercase tracking-wider text-amber-500">
              SubShield Pro Member Portal
            </span>
            <h1 className="text-2xl font-extrabold text-white">
              Sign In to Your Project Tracker
            </h1>
            <p className="text-xs text-slate-400 leading-relaxed">

            </p>

function slugifyState(name: string): string {
  return name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "");
}

  const [filterMode, setFilterMode] = useState<"all" | "prelim_required" | "no_prelim">("all");

  const allStates = useMemo(() => {
    return Object.values(STATE_LIEN_RULES).sort((a, b) =>

  const filteredStates = useMemo(() => {
    return allStates.filter((rule) => {

      const matchesSearch =

"use client";

import { useState } from "react";
import Link from "next/link";

export default function LaborBurdenCalculator() {
  // Base Wage & Hours
  const [hourlyWage, setHourlyWage] = useState<number>(32);
  const [annualHours, setAnnualHours] = useState<number>(2080); // 40 hrs * 52 wks

  // Statutory Payroll Taxes (%)
  const [ficaRate, setFicaRate] = useState<number>(7.65); // Social Security (6.2%) + Medicare (1.45%)
  const [futaRate, setFutaRate] = useState<number>(0.6); // Federal Unemployment on first $7,000
  const [sutaRate, setSutaRate] = useState<number>(3.5); // State Unemployment (varies)

  // Insurance & Benefits
  const [workersCompRate, setWorkersCompRate] = useState<number>(8.5); // % of payroll based on class code
  const [annualPaidLeaveHours, setAnnualPaidLeaveHours] = useState<number>(80); // 10 days PTO/holidays
  const [annualBenefitsCost, setAnnualBenefitsCost] = useState<number>(3600); // Health, safety gear, 401k

  // Calculations
  const grossAnnualBasePay = hourlyWage * annualHours;

  const annualFica = grossAnnualBasePay * (ficaRate / 100);
  // FUTA is capped at $7,000 max wage per employee in the US ($42 max), simplified:
  const annualFuta = Math.min(grossAnnualBasePay, 7000) * (futaRate / 100);
  const annualSuta = grossAnnualBasePay * (sutaRate / 100);
  const annualWorkersComp = grossAnnualBasePay * (workersCompRate / 100);

  const annualPaidLeaveCost = hourlyWage * annualPaidLeaveHours;
  const totalIndirectLaborCost =
    annualFica +
    annualFuta +
    annualSuta +
    annualWorkersComp +
    annualPaidLeaveCost +
    annualBenefitsCost;

  const totalTrueLaborCost = grossAnnualBasePay + totalIndirectLaborCost;
  const laborBurdenPercentage = (totalIndirectLaborCost / grossAnnualBasePay) * 100;
  const trueCostPerHour = totalTrueLaborCost / (annualHours - annualPaidLeaveHours); // True cost per actual working hour

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans">
      {/* Top Bar */}

function parseLocalDate(dateStr: string): Date | null {
  if (!dateStr) return null;
  const parts = dateStr.split("-").map(Number);
  if (parts.length !== 3 || parts.some(isNaN)) return null;
  return new Date(parts[0], parts[1] - 1, parts[2]);
}

function formatIsoDate(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

export function generateIcsFile(
  projectName: string,
  preliminaryDeadline: string | null,
  lienDeadline: string
): string {
  const formatDate = (dateStr: string) => dateStr.replace(/-/g, '');

  let events = '';

  if (preliminaryDeadline) {
    const pDate = formatDate(preliminaryDeadline);
    events += `
BEGIN:VEVENT
UID:prelim-${Date.now()}@subshield.com
DTSTAMP:${formatDate(new Date().toISOString().slice(0, 10))}T090000Z
DTSTART;VALUE=DATE:${pDate}
SUMMARY:CRITICAL: Preliminary Notice Deadline - ${projectName}
DESCRIPTION:Statutory cutoff window to deliver preliminary notice or notice to owner. Missing this deadline forfeits mechanic's lien rights.
BEGIN:VALARM
ACTION:DISPLAY
DESCRIPTION:Preliminary Notice Deadline in 5 Days!
TRIGGER:-P5D
END:VALARM
END:VEVENT`;
  }

  const lDate = formatDate(lienDeadline);
  events += `
BEGIN:VEVENT
UID:lien-${Date.now()}@subshield.com
DTSTAMP:${formatDate(new Date().toISOString().slice(0, 10))}T090000Z
DTSTART;VALUE=DATE:${lDate}
SUMMARY:FINAL CUTOFF: Mechanics Lien Filing Deadline - ${projectName}
DESCRIPTION:Last statutory date to record a formal mechanic's lien against the real property parcel.
BEGIN:VALARM
ACTION:DISPLAY
DESCRIPTION:Lien Filing Cutoff in 14 Days!
TRIGGER:-P14D
END:VALARM
END:VEVENT`;

  return `BEGIN:VCALENDAR
VERSION:2.0
PRODID:-//SubShield HQ//Lien Safe Calendar//EN
CALSCALE:GREGORIAN
METHOD:PUBLISH${events}
END:VCALENDAR`.trim();
}
